public abstract class AcDecorator: IAirConditioner{
    public IAirConditioner _Ac;
    protected AcDecorator(IAirConditioner Ac){
        _Ac = Ac;
    }
    public virtual int Temperature {get=>Ac.Temperature; set=>Ac.Temperature=value;}
    public abstract string GetDescription();
    public abstract int GetPrice();
}

Public class HepaFilterFeature: AcDecorator{
    public HepaFilterFeature(IAirConditioner ac): base(ac){

    } 

    public override string GetDescription(){
        return ac.GetDescription() + ", Hepa Filter";
    }

    public override int GetPrice(){
        return ac.GetPrice() + 5000;
    }
}

    public override string GetDescription(){
        return Ac.GetDescription() + ", With upgraded insulation";
    }
    public override int GetPrice(){
        return Ac.GetPrice() + 5000;
    }
}

Requirements

Design a parking system.

Vehicle types:

Car
Bike
Truck

Parking slots:

Small
Medium
Large

Rules:

Bike → Small slot
Car → Medium slot
Truck → Large slot

Generate ticket when vehicle enters.

Calculate fee when vehicle exits.

public enum VehicleType
{
    Bike,
    Car,
    Truck
}

=============Interface Design====================
public interface IVehicle{
    public string PlateNumber {get;}
    public VehicleType Type {get;}
}

=============Concrete Class=======================
public class Car: IVehicle{
    public string PlateNumber {get;}
    public Car(string Number){
        PlateNumber = Number;
    }
    public VehicleType Type => VehicleType.Car;
}

public class Bike: IVehicle{
    public string PlateNumber {get;}
    public Bike(string Number){
        PlateNumber = Number;
    }
    public VehicleType Type => VehicleType.Bike;
}

public Interface IParkingSlot{
    string SlotNumber {get;}
    void allotParking(IVehicle vehicle);
    decimal parkingPrice();
}

public class SmallParkingSlot: IParkingSlot{
    private IVehicle vehicle;
    public SmallParkingSlot(IVehicle vehicle){
        _vehicle = new Bike(vehicle.Number, vehicle.Type);
    }
    public void allotParking(IVehicle vehicle){
        this.vehicle = vehicle;
    }
    public double ParkingPrice(){
        return 100;
    }
}

public class MediumParkingSlot: IParkingSlot{
    IVehicle _vehicle;
    public MediumParkingSlot(IVehicle vehicle){
        _vehicle = new Car(vehicle.Number, vehicle.Type);
    }
    public string allotParking(){
        return "Medium Slot";
    }
    public double ParkingPrice(){
        return 150;
    }
}

public class LargeParkingSlot: IParkingSlot{
    IVehicle _vehicle;
    public LargeParkingSlot(IVehicle vehicle){
        _vehicle = new Truck(vehicle.Number, vehicle.Type);
    }
    public string allotParking(){
        return "Large Slot";
    }
    public double ParkingPrice(){
        return 200;
    }
}

==================Factory Class=======================
public class ParkingSlotFactory{
    IVehicle _vehicle;
    public ParkingSlotFactory(IVehicle vehicle){
        _vehicle = vehicle;
    }
    public IParkingSlot buildParkingSlot(){
        if(_vehicle.VehicleType == "Bike"){
            return new SmallParkingSlot();
        } else if(_vehicle.VehicleType == "Car"){
            return new MediumParkingSlot();
        } else{
            return new LargeParkingSlot();
        }
    }
}

public class ParkingManager{
    IVehicle _vehicle {get; set;}
    public ParkingSlotFactory _factory;

    public ParkingManager(IVehicle vehicle){
        _vehicle = vehicle;
    }

    public ParkingResponse BookParking(){
        _factory = new ParkingSlotFactory(_vehicle);
        IParkingSlot bookedslot = _factory.buildParkingSlot();
        return new ParkingResponse {
            slot = bookedslot.allotParking(),
            vehicletype = _vehicle.VehicleType;
            plateNumber = _vehicle.PlateNumber;
            price = bookedslot.ParkingPrice();
        }
    }
}

public class ParkingResponse{
    string slot {get; set;}
    string vehicletype {get; set;}
    string plateNumer {get; set;}
    double price {get; set;}
}

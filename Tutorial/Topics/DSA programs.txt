Stack
=====================
Stack data = new Stack<string>();
data.Push("Apple");
data.Push("Orange");
data.Push("Banana");
data.Pop(); // LIFO Banana

foreach(string fruit in data){
    Console.WriteLine(fruit); //  Orange, Apple
    // it will loop from top to bottom, means last in will print first.
}

To print from Bottom to top
string[] arr = data.ToArray();
for(let i=Length-1; i>=0; i--){
    Console.WriteLine(arr[i]);
}
